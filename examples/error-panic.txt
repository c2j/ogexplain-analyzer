
SELECT -1 * NVL(SUM(T.DGH_VALUE), 0) / 100
               INTO V_EXT_QUANTITY
               FROM V_DAT_TRD_PROXY_BALANCE T,
                    V_PAR_INVEST_ACNT_INFO  N,
                    PAR_SYS_SECURITIES      V
              WHERE T.BCRQ = v_deal_date
                AND T.MEMBER_ACNT = N.INVEST_ACNT
                AND N.ACNT_ID = R_SECU_DTL.ACCOUNT_ID
                AND V.SECURITY_ID = R_SECU_DTL.SECURITY_ID
                AND T.PRODUCT_CODE = DECODE(V.MAIN_MARKET_CODE,GLOBAL_TSH_MARKET_CODE,V.STOCK_CODE||'SH',
                GLOBAL_TSZ_MARKET_CODE,V.STOCK_CODE||'SZ',V.STOCK_CODE)
                AND V.STOCK_STATUS = '00' --上市流通
                AND V.MAIN_MARKET_CODE IN ('003', '046',GLOBAL_TSH_MARKET_CODE,GLOBAL_TSZ_MARKET_CODE) --上清所
                AND v_deal_date BETWEEN V.INURE_BEGIN_DATE AND V.INURE_END_DATE;

FROM dat_zl_accounthistoryresult a
LEFT JOIN par_fund_info b ON a.fund_id = b.fund_id
LEFT JOIN par_sys_coin c ON a.coin_code = c.coin_code
WHERE a.send_recv = '1'
AND a.fund_id = p_i_fund_id
AND a.begin_date = p_i_data_date
AND b.fund_code IN (SELECT fund_code
FROM v_fundcode_priv
WHERE user_id = i_user_id
AND role = i_role_id)
        and rownum <=10000
        ORDER BY a.seq_code,
